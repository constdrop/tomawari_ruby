### サンプルコード

```ruby
# キャラクターの基底クラスを定義する
class Character
  attr_accessor :name, :hp, :attack, :level, :exp

  def initialize(name, hp, attack)
    @name = name
    @hp = hp
    @attack = attack
    @level = 1
    @exp = 0
  end

  def attack(target)
    damage = @attack
    target.hp -= damage
    puts "#{@name}の攻撃！#{target.name}に#{damage}のダメージを与えた！"
    if target.hp <= 0
      puts "#{target.name}を倒した！"
    else
      puts "#{target.name}の残りHPは#{target.hp}だ！"
    end
  end

  def defend
    puts "#{@name}は防御した！"
  end
end

# プレイヤークラスを定義する(Characterクラスを継承)
class Player < Character
  def initialize(name)
    super(name, 100, 10)
  end

  def gain_exp(exp)
    @exp += exp
    puts "#{@name}は#{exp}の経験値を獲得した！"
    level_up
  end

  def level_up
    while @exp >= @level * 10  # 必要な経験値はレベル * 10
      @exp -= @level * 10
      @level += 1
      @hp += 10
      @attack += 2
      puts "#{@name}はレベル#{@level}に上がった！"
      puts "HPが#{@hp}、攻撃力が#{@attack}になった！"
    end
  end
end

# 敵クラスを定義する(Characterクラスを継承)
class Enemy < Character
  attr_reader :exp

  def initialize(name, hp, attack, exp)
    super(name, hp, attack)
    @exp = exp
  end
end

# ゲームを開始する
puts "冒険が始まる！"
puts "主人公の名前を入力してください："
player_name = gets.chomp
player = Player.new(player_name)

# ステージデータ
stages = [
  {
    name: "草原",
    enemies: [
      Enemy.new("スライム", 50, 5, 10),
      Enemy.new("ゴブリン", 70, 8, 20),
      Enemy.new("オーク", 90, 10, 30)
    ]
  },
  {
    name: "森",
    enemies: [
      Enemy.new("オオカミ", 80, 12, 40),
      Enemy.new("スケルトン", 100, 15, 50),
      Enemy.new("ゴースト", 120, 18, 60)
    ]
  },
  {
    name: "洞窟",
    enemies: [
      Enemy.new("ドラゴン", 150, 20, 70),
      Enemy.new("ゴーレム", 180, 25, 80),
      Enemy.new("魔王", 200, 30, 100)
    ]
  }
]

# 各ステージを順番にプレイ
stages.each do |stage|
  puts "--------------------"
  puts "#{stage[:name]}に到着！"
  puts "--------------------"

  stage[:enemies].each do |enemy|
    puts "#{enemy.name}が現れた！"

    # 戦闘開始
    while player.hp > 0 && enemy.hp > 0
      puts "--------------------"
      puts "#{player.name}のHP：#{player.hp}"
      puts "#{enemy.name}のHP：#{enemy.hp}"
      puts "--------------------"
      puts "コマンド？"
      puts "1. 攻撃"
      puts "2. 防御"
      puts "3. 逃げる"
      command = gets.chomp.to_i

      case command
      when 1
        player.attack(enemy)
        if enemy.hp > 0
          enemy.attack(player)
        else
          player.gain_exp(enemy.exp) # 敵を倒したら経験値を獲得
        end
      when 2
        player.defend
        enemy.attack(player)
      when 3
        puts "逃げることに成功した！"
        break
      else
        puts "無効なコマンドです"
      end
    end

    if player.hp <= 0
      puts "#{player.name}は力尽きた..."
      puts "ゲームオーバー"
      exit
    end
  end

  puts "#{stage[:name]}をクリア！"
end

puts "全てのステージをクリア！ゲームクリア！"
```
