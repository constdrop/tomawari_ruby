require 'json'

def slow_puts(text, speed = 0.05)
  text.each_char do |char|
    print char
    sleep speed
    $stdout.flush # バッファをクリア
  end
  puts # 改行
end

# キャラクターの基底クラスを定義する
class Character
  attr_accessor :name, :hp, :power, :level, :exp, :critical_rate

  def initialize(name, hp, power, critical_rate = 0.05)
    @name = name
    @hp = hp
    @power = power
    @level = 1
    @exp = 0
    @critical_rate = critical_rate
  end

  def attack(target)
    damage = @power
    if rand < @critical_rate # クリティカルヒット判定
      damage *= 2 # ダメージを2倍にする
      puts "クリティカルヒット！"
    end

    target.hp -= damage
    slow_puts "#{@name}の攻撃！#{target.name}に#{damage}のダメージを与えた！"
    if target.hp <= 0
      slow_puts "#{target.name}を倒した！"
    else
      slow_puts "#{target.name}の残りHPは#{target.hp}だ！"
    end
  end

  def defend
    slow_puts "#{@name}は防御した！"
  end
end

# プレイヤークラスを定義する(Characterクラスを継承)
class Player < Character
  attr_accessor :is_defending, :inventory, :magic_list, :max_hp, :mp, :max_mp

  def initialize(name)
    super(name, 100, 10, 0.1) # クリティカルヒット率を10%に設定
    @inventory = []
    @magic_list = [] # 魔法リストを追加、初期状態は空のリスト
    @max_hp = 100
    @mp = 50        # MPを追加
    @max_mp = 50    # 最大MPを追加
    @is_defending = false # 初期状態は防御していない
  end

  def gain_exp(exp)
    @exp += exp
    slow_puts "#{@name}は#{exp}の経験値を獲得した！"
    level_up
  end

  def level_up
    while @exp >= @level * 10  # 必要な経験値はレベル * 10
      @exp -= @level * 10
      @level += 1
      @hp += 10
      @power += 2
      @max_mp += 20  # 最大MPを追加
      @mp += 10       # MPを追加
      slow_puts "#{@name}はレベル#{@level}に上がった！"
      slow_puts "HPが#{@hp}、攻撃力が#{@power}になった！"
      slow_puts "最大MPが#{@max_mp}に、MPが#{@mp}になった！"

      # 一定のレベル達したら魔法を覚える
      case @level
      when 10
        player.magic_list << HealingMagic.new("ヒーラ", 15, 50)
        slow_puts "ヒーラの魔法を覚えた！"
      end
    end
  end

  def defend
    super # 親クラスのdefendメソッドを呼び出す
    @is_defending = true
  end

  # 攻撃を受けた時のメソッド
  def take_damage(damage)
    if @is_defending
      damage /= 2 # 防御時はダメージを半減
      @is_defending = false # 防御状態を解除
    end
    @hp -= damage
    puts "#{@name}は#{damage}のダメージを受けた！"
  end
end

# 敵クラスを定義する(Characterクラスを継承)
class Enemy < Character
  attr_reader :exp

  def initialize(name, hp, power, exp, critical_rate = 0.05) # クリティカルヒット率を引数で指定
    super(name, hp, power, critical_rate)
    @exp = exp
  end
end

class Item
  attr_reader :name

  def initialize(name)
    @name = name
  end

  def use(target, user)
    # アイテムの使用効果は子クラスで定義する
    raise NotImplementedError, "子クラスでこのメソッドをオーバーライドしてください"
  end
end

class HealingItem < Item
  attr_reader :effect

  def initialize(name, effect)
    super(name)
    @effect = effect
  end

  def use(target, user)
    heal_amount = @effect
    target.hp += heal_amount
    slow_puts "#{target.name}は#{@name}を使ってHPが#{heal_amount}回復した！"
    # 最大HPを超えないように調整
    target.hp = [target.hp, target.max_hp].min if target.max_hp
    
    user.inventory.delete(self)
    puts "#{user.name}は#{name}を使った。"
  end
end

class Magic
  attr_reader :name, :mp_cost

  def initialize(name, mp_cost)
    @name = name
    @mp_cost = mp_cost
  end

  def use(caster, target)
    # 魔法の使用効果は子クラスで定義する
    raise NotImplementedError, "子クラスでこのメソッドをオーバーライドしてください"
  end
end

class HealingMagic < Magic
  attr_reader :effect

  def initialize(name, mp_cost, effect)
    super(name, mp_cost)
    @effect = effect
  end

  def use(caster, target)
    if caster.mp >= @mp_cost
      caster.mp -= @mp_cost
      heal_amount = @effect
      target.hp += heal_amount
      slow_puts "#{caster.name}は#{@name}を唱え、#{target.name}のHPを#{heal_amount}回復した！"
      # 最大HPを超えないように調整
      target.hp = [target.hp, target.max_hp].min if target.max_hp
    else
      slow_puts "#{caster.name}はMPが足りない！"
    end
  end
end

def create_save_data(player, current_stage_index)
  {
    player: {
      name: player.name,
      hp: player.hp,
      max_hp: player.max_hp,
      power: player.power,
      level: player.level,
      exp: player.exp,
      mp: player.mp,
      max_mp: player.max_mp,
      inventory: player.inventory.map { |item| { name: item.name, class: item.class.to_s, effect: item.respond_to?(:effect) ? item.effect : nil } },
      magic_list: player.magic_list.map { |magic| { name: magic.name, mp_cost: magic.mp_cost, class: magic.class.to_s, effect: magic.respond_to?(:effect) ? magic.effect : nil } }
    },
    current_stage_index: current_stage_index
  }
end

def save_game(player, current_stage_index, filename = "save_data.json")
  save_data = create_save_data(player, current_stage_index)
  File.open(filename, "w") do |file|
    JSON.dump(save_data, file)
  end
  puts "ゲームをセーブしました。"
end

def load_game(filename = "save_data.json")
  if File.exist?(filename)
    File.open(filename, "r") do |file|
      save_data = JSON.load(file)
      player_data = save_data["player"]
      current_stage_index = save_data["current_stage_index"]

      player = Player.new(player_data["name"])
      player.hp = player_data["hp"]
      player.max_hp = player_data["max_hp"]
      player.power = player_data["power"]
      player.level = player_data["level"]
      player.exp = player_data["exp"]
      player.mp = player_data["mp"]
      player.max_mp = player_data["max_mp"]

      player.inventory = player_data["inventory"].map do |item_data|
        item_class = Object.const_get(item_data["class"])
        item_class.new(item_data["name"], item_data["effect"])
      end

      player.magic_list = player_data["magic_list"].map do |magic_data|
        magic_class = Object.const_get(magic_data["class"])
        magic_class.new(magic_data["name"], magic_data["mp_cost"], magic_data["effect"])
      end

      puts "ゲームをロードしました。"
      return player, current_stage_index
    end
  else
    puts "セーブデータが見つかりません。"
    return nil, 0
  end
end

end

if current_stage_index == -1

stages = [

# 各ステージを順番にプレイ
stages.each_with_index do |stage, stage_index|
  # ロードしたステージまでは何もしないで次のステージへ
  next if stage_index < current_stage_index

  puts "--------------------"
  slow_puts "#{stage[:name]}に到着！"
  puts "--------------------"

  stage[:enemies].each do |enemy|
    slow_puts "#{enemy.name}が現れた！"

    # 戦闘開始
    while player.hp > 0 && enemy.hp > 0
      puts "--------------------"
      slow_puts "#{player.name}のHP：#{player.hp}, MP：#{player.mp}"
      slow_puts "#{enemy.name}のHP：#{enemy.hp}"
      puts "--------------------"
      puts "コマンド？"
      puts "1. 攻撃"
      puts "2. 防御"
      puts "3. 逃げる"
      puts "4. アイテム"
      puts "5. 魔法" # 魔法コマンドを追加
      command = gets.chomp.to_i

      case command
      when 1
        player.attack(enemy)
        if enemy.hp > 0
          enemy.attack(player)
        else
          player.gain_exp(enemy.exp) # 敵を倒したら経験値を獲得
          # 10回に1回の確率でやくそうを獲得
          if Random.rand(10) == 0
            player.inventory << HealingItem.new("やくそう", 20)
          end
        end
      when 2
        player.defend
        enemy.attack(player)
      when 3
        slow_puts "逃げることに成功した！"
        break
      when 4 # アイテムの場合
        slow_puts "どのアイテムを使う？"
        player.inventory.each_with_index do |item, index|
          puts "#{index + 1}. #{item.name}"
        end
        item_index = gets.chomp.to_i - 1
        if item = player.inventory[item_index]
          item.use(player, player) # プレイヤーを引数として渡す
        else
          puts "無効なアイテムです。"
        end
      when 5 # 魔法の場合
        slow_puts "どの魔法を使う？"
        player.magic_list.each_with_index do |magic, index|
          puts "#{index + 1}. #{magic.name} (MP: #{magic.mp_cost})"
        end
        magic_index = gets.chomp.to_i - 1
        if magic = player.magic_list[magic_index]
          magic.use(player, player)
        else
          puts "無効な魔法です。"
        end
      else
        puts "無効なコマンドです"
      end
    end

    if player.hp <= 0
      slow_puts "#{player.name}は力尽きた..."
      slow_puts "ゲームオーバー"
      exit
    end
  end

  slow_puts "#{stage[:name]}をクリア！"
  save_game(player, stage_index + 1) # ステージクリア時に自動でセーブ
end

slow_puts "全てのステージをクリア！ゲームクリア！"
